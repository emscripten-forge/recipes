cmake_minimum_required(VERSION 4.3.3)
project(check_libtorch NONE)
find_package(Torch REQUIRED CONFIG)
find_package(ATen REQUIRED CONFIG)
foreach(header IN ITEMS ATen/ATen.h torch/csrc/api/include/torch/torch.h)
  if(NOT EXISTS "${TORCH_INSTALL_PREFIX}/include/${header}")
    message(FATAL_ERROR "Missing libtorch header: ${header}")
  endif()
endforeach()
if(NOT EXISTS "${ATEN_INCLUDE_DIR}/ATen/ATen.h")
  message(FATAL_ERROR "ATen include directory is not relocatable")
endif()
