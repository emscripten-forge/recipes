cmake_minimum_required(VERSION 3.20)
project(test_clang_static LANGUAGES C CXX)

find_package(LLVM $ENV{PKG_VERSION} EXACT REQUIRED CONFIG)
if(PACKAGE_NAME STREQUAL "clang-interpreter-static")
  find_package(LLD $ENV{PKG_VERSION} EXACT REQUIRED CONFIG)
endif()
find_package(Clang $ENV{PKG_VERSION} EXACT REQUIRED CONFIG)

set(required_targets clangBasic clangFrontend)
if(PACKAGE_NAME STREQUAL "clang-interpreter-static")
  list(APPEND required_targets clangInterpreter lldCommon lldWasm)
elseif(TARGET clangInterpreter OR TARGET lldCommon OR TARGET lldWasm)
  message(FATAL_ERROR "Core Clang must be installable independently of Interpreter and LLD")
endif()

foreach(component libclang clang-cpp)
  if(TARGET ${component})
    message(FATAL_ERROR "The static package must not export the shared target: ${component}")
  endif()
endforeach()
foreach(component ${required_targets})
  if(NOT TARGET ${component})
    message(FATAL_ERROR "Missing imported target: ${component}")
  endif()
endforeach()
